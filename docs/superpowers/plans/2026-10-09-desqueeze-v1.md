```swift
import Foundation

/// A horizontal stretch factor for the DNG `DefaultScale` tag.
///
/// The tag holds two numbers: horizontal stretch, then vertical stretch. This
/// app writes `<factor> 1`. A factor of 1 is no stretch.
public struct SqueezeFactor: Equatable, Hashable, Sendable {
    public static let range: ClosedRange<Double> = 1...4
    /// How close a read-back value must be to count as this factor: less than
    /// half of the smallest step between two factors, so neighbors never match.
    public static let tolerance = 0.00005
    public static let presets: [SqueezeFactor] = [1.33, 1.5, 1.8, 2].compactMap { SqueezeFactor($0) }
    /// No stretch. Remove writes this. Not named `none`: on a `SqueezeFactor?`,
    /// `.none` would mean nil.
    public static let noStretch = SqueezeFactor(1)!

    public let value: Double

    /// Rounds to four decimal places, then checks the range.
    public init?(_ value: Double) {
        let rounded = (value * 10_000).rounded() / 10_000
        guard Self.range.contains(rounded) else { return nil }
        self.value = rounded
    }

    /// Parses what the user typed: digits, and at most four decimal places.
    public init?(text: String) {
        let trimmed = text.trimmingCharacters(in: .whitespaces)
        guard trimmed.range(of: #"^[0-9]+(\.[0-9]{1,4})?$"#, options: .regularExpression) != nil,
              let number = Double(trimmed) else { return nil }
        self.init(number)
    }

    /// The factor with no trailing zeros: `1.5`, `2`, `1.3333`.
    public var label: String {
        var text = String(format: "%.4f", value)
        while text.hasSuffix("0") { text.removeLast() }
        if text.hasSuffix(".") { text.removeLast() }
        return text
    }

    /// The value exiftool writes: `<factor> 1`.
    public var tagValue: String { "\(label) 1" }

    public func matches(horizontal: Double, vertical: Double) -> Bool {
        abs(horizontal - value) < Self.tolerance && abs(vertical - 1) < Self.tolerance
    }
}
```

```swift
        let arguments: [String]
```

```swift
import Foundation

public struct ProcessResult: Equatable, Sendable {
    public let exitCode: Int32
    public let stdout: String
    public let stderr: String

    public init(exitCode: Int32, stdout: String, stderr: String) {
        self.exitCode = exitCode
        self.stdout = stdout
        self.stderr = stderr
    }
}

/// Starts a program with arguments and waits for it. No shell is involved, so
/// a path with spaces is one argument.
public protocol ProcessRunner: Sendable {
    func run(_ executable: URL, _ arguments: [String]) throws -> ProcessResult
}

public struct SystemProcessRunner: ProcessRunner {
    public init() {}

    public func run(_ executable: URL, _ arguments: [String]) throws -> ProcessResult {
        // Foundation autoreleases the pipe handles. Without this pool, a caller
        // that loops holds two descriptors for each call until its own pool drains.
        try autoreleasepool {
            let process = Process()
            process.executableURL = executable
            process.arguments = arguments
            let out = Pipe()
            let err = Pipe()
            process.standardOutput = out
            process.standardError = err
            process.standardInput = FileHandle.nullDevice
            try process.run()

            // Read both pipes at the same time. A program that fills one pipe
            // while this code waits on the other would block forever. A serial
            // queue always gets a thread of its own, also when every thread of
            // the Swift concurrency pool is blocked in this method.
            let errData = DataBox()
            let group = DispatchGroup()
            DispatchQueue(label: "org.plateworks.desqueeze.stderr").async(group: group) {
                errData.value = err.fileHandleForReading.readDataToEndOfFile()
            }
            let outData = out.fileHandleForReading.readDataToEndOfFile()
            group.wait()
            process.waitUntilExit()

            return ProcessResult(
                exitCode: process.terminationStatus,
                stdout: String(decoding: outData, as: UTF8.self),
                stderr: String(decoding: errData.value, as: UTF8.self)
            )
        }
    }
}

private final class DataBox: @unchecked Sendable {
    var value = Data()
}
```

```swift
import Foundation

/// Finds the two external tools. Holds no path that is specific to one Mac.
///
/// A Mac app started from Finder gets a short `PATH` with no Homebrew folder,
/// so the two fixed Homebrew locations come first.
public struct ToolLocator: Sendable {
    public static let converterBundleID = "com.adobe.DNGConverter"
    public static let converterBinary = "/Contents/MacOS/Adobe DNG Converter"
    public static let standardConverterApp = "/Applications/Adobe DNG Converter.app"

    public let exiftoolCandidates: [String]
    public let converterApps: [String]
    private let isExecutable: @Sendable (String) -> Bool

    /// - Parameter registeredConverterApp: the app that Launch Services
    ///   returns for `converterBundleID`. The app layer supplies it.
    public init(
        pathVariable: String? = ProcessInfo.processInfo.environment["PATH"],
        registeredConverterApp: URL? = nil,
        isExecutable: @escaping @Sendable (String) -> Bool = { FileManager.default.isExecutableFile(atPath: $0) }
    ) {
        var tools = ["/opt/homebrew/bin/exiftool", "/usr/local/bin/exiftool"]
        // A relative entry would resolve against the app's working folder.
        for folder in (pathVariable ?? "").split(separator: ":") where folder.hasPrefix("/") {
            let candidate = "\(folder)/exiftool"
            if !tools.contains(candidate) { tools.append(candidate) }
        }
        exiftoolCandidates = tools

        var apps: [String] = []
        if let registered = registeredConverterApp { apps.append(registered.path) }
        if !apps.contains(Self.standardConverterApp) { apps.append(Self.standardConverterApp) }
        converterApps = apps
        self.isExecutable = isExecutable
    }

    public func exiftool() -> URL? {
        exiftoolCandidates.first(where: isExecutable).map { URL(fileURLWithPath: $0) }
    }

    public func dngConverter() -> URL? {
        converterApps.map { $0 + Self.converterBinary }
            .first(where: isExecutable)
            .map { URL(fileURLWithPath: $0) }
    }
}
```

```swift
    public var dngs: [URL]
    public var arws: [URL]

public enum FolderScan {
    /// The DNG and ARW files at the top level of `folder`, sorted by name.
    /// Subfolders, hidden files, and other file types are not returned.
    public static func contents(of folder: URL) throws -> FolderContents {
        let items = try FileManager.default.contentsOfDirectory(
            at: folder, includingPropertiesForKeys: [.isRegularFileKey], options: [])
        var contents = FolderContents(dngs: [], arws: [])
        for item in items {
            guard !item.lastPathComponent.hasPrefix("."),
                  (try? item.resourceValues(forKeys: [.isRegularFileKey]).isRegularFile) == true
            else { continue }
            switch item.pathExtension.lowercased() {
            case "dng": contents.dngs.append(item)
            case "arw": contents.arws.append(item)
            default: break
            }
        }
        let byName: (URL, URL) -> Bool = {
            $0.lastPathComponent.localizedStandardCompare($1.lastPathComponent) == .orderedAscending
        }
        contents.dngs.sort(by: byName)
        contents.arws.sort(by: byName)
        return contents
    }

    /// The ARW files that no DNG claims.
    ///
    /// A DNG claims an ARW when the two have the same base name, or when the
    /// DNG's `originalRawFileName` equals the ARW's file name. Letter case is
    /// ignored. The second rule protects a folder whose DNG files were renamed
    /// on import: without it, a rerun would convert each ARW again.
    ///
    /// Either claim blocks conversion, on purpose. The converter never
    /// overwrites, so a wrong "pending" writes a `_1.dng` duplicate, and a
    /// wrong "claimed" only leaves one ARW waiting. Known limit: a DNG that
    /// was renamed to another shot's base name claims that shot's ARW too.
    public static func pending(arws: [URL], dngs: [DNGFile]) -> [URL] {
        var claimedBaseNames = Set<String>()
        var claimedFileNames = Set<String>()
        for dng in dngs {
            claimedBaseNames.insert(dng.url.deletingPathExtension().lastPathComponent.lowercased())
            if let original = dng.originalRawFileName { claimedFileNames.insert(original.lowercased()) }
        }
        return arws.filter { arw in
            !claimedBaseNames.contains(arw.deletingPathExtension().lastPathComponent.lowercased())
                && !claimedFileNames.contains(arw.lastPathComponent.lowercased())
        }
    }
}
```

```swift
import Foundation

/// One row of the folder read: a DNG and its current tag.
public struct TagRow: Equatable, Sendable {
    public let fileName: String
    public let horizontal: Double
    public let vertical: Double
    public let originalRawFileName: String?
}

public enum TagError: Error, Equatable {
    case toolFailed(String)
}

/// Reads and writes the `DefaultScale` tag. `DesqueezeJob` depends on this
/// protocol, so its tests need no exiftool.
public protocol ScaleTagging: Sendable {
    func readFolder(_ folder: URL) throws -> [TagRow]
    func write(_ factor: SqueezeFactor, to file: URL) throws
    func read(_ file: URL) throws -> (horizontal: Double, vertical: Double)?
}

/// The exiftool implementation.
///
/// The group `SubIFD` is named on every call, because that is the image
/// directory Camera Raw reads.
public struct ScaleTag: ScaleTagging {
    static let tagName = "-SubIFD:DefaultScale"

    private let exiftool: URL
    private let runner: ProcessRunner

    public init(exiftool: URL, runner: ProcessRunner = SystemProcessRunner()) {
        self.exiftool = exiftool
        self.runner = runner
    }

    /// One call for the whole folder. exiftool reads the top level only.
    public func readFolder(_ folder: URL) throws -> [TagRow] {
        let result = try runner.run(exiftool, [
            "-m", "-T", "-filename", Self.tagName, "-OriginalRawFileName", "-ext", "dng", folder.path,
        ])
        return Self.parseRows(result.stdout)
    }

    /// Writes the tag. exiftool keeps the previous file as `<name>_original`.
    /// `-m` is needed: Adobe DNG Converter files carry a minor flaw in a Sony
    /// data block, and exiftool refuses to write without it.
    public func write(_ factor: SqueezeFactor, to file: URL) throws {
        let result = try runner.run(exiftool, ["-m", "\(Self.tagName)=\(factor.tagValue)", file.path])
        guard result.exitCode == 0, result.stdout.contains("1 image files updated") else {
            throw TagError.toolFailed(Self.detail(result))
        }
    }

    public func read(_ file: URL) throws -> (horizontal: Double, vertical: Double)? {
        let result = try runner.run(exiftool, ["-s3", Self.tagName, file.path])
        return Self.parseScale(result.stdout)
    }

    /// Parses `1.5 1`. Returns nil for `-`, an empty string, or anything else.
    static func parseScale(_ text: String) -> (horizontal: Double, vertical: Double)? {
        let parts = text.split(whereSeparator: { $0 == " " || $0 == "\n" || $0 == "\t" })
        guard parts.count == 2, let horizontal = Double(parts[0]), let vertical = Double(parts[1]) else {
            return nil
        }
        return (horizontal, vertical)
    }

    /// Parses the tab-separated folder read. exiftool includes hidden files,
    /// so this drops each row whose name starts with a period.
    static func parseRows(_ text: String) -> [TagRow] {
        text.split(separator: "\n").compactMap { line in
            let cells = line.split(separator: "\t", omittingEmptySubsequences: false).map(String.init)
            guard cells.count == 3, !cells[0].hasPrefix(".") else { return nil }
            let scale = parseScale(cells[1]) ?? (1, 1)
            return TagRow(fileName: cells[0], horizontal: scale.horizontal, vertical: scale.vertical,
                          originalRawFileName: cells[2] == "-" ? nil : cells[2])
        }
    }

    /// One line that says why exiftool failed. The hidden-data warning is
    /// expected on every Adobe DNG Converter file and says nothing.
    static func detail(_ result: ProcessResult) -> String {
        let lines = (result.stderr + "\n" + result.stdout)
            .split(separator: "\n")
            .map { $0.trimmingCharacters(in: .whitespaces) }
            .filter { !$0.isEmpty && !$0.contains("Error copying hidden data") }
        return lines.first(where: { $0.hasPrefix("Error") }) ?? lines.last
            ?? "exiftool returned \(result.exitCode)."
    }
}
```

```swift
public protocol BackupHandling: Sendable {
    func backupExists(for dng: URL) -> Bool
    /// Moves the backup to the Trash.
    func trashBackup(for dng: URL) throws
    /// Replaces the DNG with its backup. The backup file is consumed.
    func restoreBackup(for dng: URL) throws
}

public struct FileBackups: BackupHandling {
    private let trash: @Sendable (URL) throws -> Void

    /// - Parameter trash: how to discard a backup. The default moves it to the
    ///   Trash. Tests inject a function, so they never fill the real Trash.
    public init(trash: @escaping @Sendable (URL) throws -> Void = {
        try FileManager.default.trashItem(at: $0, resultingItemURL: nil)
    }) {
        self.trash = trash
    }

    public static func backupURL(for dng: URL) -> URL {
        URL(fileURLWithPath: dng.path + "_original")
    }

    public func backupExists(for dng: URL) -> Bool {
        FileManager.default.fileExists(atPath: Self.backupURL(for: dng).path)
    }

    public func trashBackup(for dng: URL) throws {
        try trash(Self.backupURL(for: dng))
    }

    public func restoreBackup(for dng: URL) throws {
        _ = try FileManager.default.replaceItemAt(dng, withItemAt: Self.backupURL(for: dng))
```

```swift
import Foundation

public enum ConvertError: Error, Equatable {
    case failed(String)
}

/// Converts one raw file to a DNG beside it and returns the DNG.
///
/// Version 1 has one implementation. A dnglab implementation is the planned
/// replacement for a public release, because Adobe's converter cannot be
/// bundled (design, section 3).
public protocol RawConverter: Sendable {
    func convert(_ arw: URL) throws -> URL
}

public struct AdobeConverter: RawConverter {
    private let executable: URL
    private let runner: ProcessRunner
    private let fileExists: @Sendable (String) -> Bool

    public init(
        executable: URL,
        runner: ProcessRunner = SystemProcessRunner(),
        fileExists: @escaping @Sendable (String) -> Bool = { FileManager.default.fileExists(atPath: $0) }
    ) {
        self.executable = executable
        self.runner = runner
        self.fileExists = fileExists
    }

    public static func outputURL(for arw: URL) -> URL {
        arw.deletingPathExtension().appendingPathExtension("dng")
    }

    /// `-c` is lossless compression. `-d` is the output folder. The command
    /// line ignores the converter's window preferences.
    ///
    /// Call this only for a pending ARW. When `<base>.dng` exists already, the
    /// converter writes `<base>_1.dng`, and this method reports success for
    /// the old file.
    ///
    /// Never pass `-h`: it opens a window and the process never returns.
    public func convert(_ arw: URL) throws -> URL {
        let output = Self.outputURL(for: arw)
        let result = try runner.run(executable, ["-c", "-d", arw.deletingLastPathComponent().path, arw.path])
        guard result.exitCode == 0 else {
            throw ConvertError.failed("Adobe DNG Converter returned \(result.exitCode).")
        }
        // The converter returns 0 for a file it cannot read, and makes no DNG.
        guard fileExists(output.path) else {
            throw ConvertError.failed("Adobe DNG Converter made no DNG. It may not support this file.")
        }
        return output
    }
}
```

```swift
/// An ordered record of what the fakes were asked to do.
final class CallLog: @unchecked Sendable {
    var entries: [String] = []
}

final class FakeBackups: BackupHandling, @unchecked Sendable {
    let log: CallLog
    var existing: Set<String> = []
    var trashError: Error?
    var restoreError: Error?

    init(log: CallLog) { self.log = log }

    func backupExists(for dng: URL) -> Bool { existing.contains(dng.lastPathComponent) }

    func trashBackup(for dng: URL) throws {
        log.entries.append("trash \(dng.lastPathComponent)")
        if let trashError { throw trashError }
        existing.remove(dng.lastPathComponent)
    }

    func restoreBackup(for dng: URL) throws {
        log.entries.append("restore \(dng.lastPathComponent)")
        if let restoreError { throw restoreError }
        existing.remove(dng.lastPathComponent)
    }
}

/// Acts like exiftool: a good write leaves a backup, a bad write leaves none.
final class FakeTag: ScaleTagging, @unchecked Sendable {
    let log: CallLog
    let backups: FakeBackups
    var rows: [TagRow] = []
    var failWrites: Set<String> = []
    /// File names whose read-back does not match what was written.
    var wrongReadBack: Set<String> = []
    var onWrite: (() -> Void)?
    private var written: [String: SqueezeFactor] = [:]

    init(log: CallLog, backups: FakeBackups) {
        self.log = log
        self.backups = backups
    }

    func readFolder(_ folder: URL) throws -> [TagRow] {
        log.entries.append("readFolder")
        return rows
    }

    func write(_ factor: SqueezeFactor, to file: URL) throws {
        let name = file.lastPathComponent
        log.entries.append("write \(factor.tagValue) \(name)")
        onWrite?()
        if failWrites.contains(name) { throw TagError.toolFailed("Error: write failed") }
        written[name] = factor
        backups.existing.insert(name)
    }

    func read(_ file: URL) throws -> (horizontal: Double, vertical: Double)? {
        let name = file.lastPathComponent
        log.entries.append("read \(name)")
        if wrongReadBack.contains(name) { return (1, 1) }
        return written[name].map { ($0.value, 1) }
    }
}

final class FakeConverter: RawConverter, @unchecked Sendable {
    let log: CallLog
    var fail: Set<String> = []

    init(log: CallLog) { self.log = log }

    func convert(_ arw: URL) throws -> URL {
        log.entries.append("convert \(arw.lastPathComponent)")
        if fail.contains(arw.lastPathComponent) { throw ConvertError.failed("made no DNG") }
        return AdobeConverter.outputURL(for: arw)
    }
}
```

```swift
import Foundation

/// Everything the window and the job need to know about one folder.
public struct FolderSnapshot: Equatable, Sendable {
    public let folder: URL
    public var dngs: [DNGFile]
    /// The ARW files that no DNG claims. Only these may go to the converter.
    public var pendingARWs: [URL]
    public var isWritable: Bool

    public var photoCount: Int { dngs.count + pendingARWs.count }
    public var isEmpty: Bool { photoCount == 0 }

    public init(folder: URL, dngs: [DNGFile], pendingARWs: [URL], isWritable: Bool) {
        self.folder = folder
        self.dngs = dngs
        self.pendingARWs = pendingARWs
        self.isWritable = isWritable
    }
}

public struct FolderScanner: Sendable {
    private let tag: ScaleTagging
    private let backups: BackupHandling

    public init(tag: ScaleTagging, backups: BackupHandling = FileBackups()) {
        self.tag = tag
        self.backups = backups
    }

    public func snapshot(of folder: URL) throws -> FolderSnapshot {
        let contents = try FolderScan.contents(of: folder)
        var rows: [String: TagRow] = [:]
        if !contents.dngs.isEmpty {
            for row in try tag.readFolder(folder) { rows[row.fileName] = row }
        }
        let dngs = contents.dngs.map { url -> DNGFile in
            let row = rows[url.lastPathComponent]
            return DNGFile(
                url: url,
                horizontal: row?.horizontal ?? 1,
                vertical: row?.vertical ?? 1,
                originalRawFileName: row?.originalRawFileName,
                hasBackup: backups.backupExists(for: url)
            )
        }
        return FolderSnapshot(
            folder: folder,
            dngs: dngs,
            pendingARWs: FolderScan.pending(arws: contents.arws, dngs: dngs),
            isWritable: FileManager.default.isWritableFile(atPath: folder.path)
        )
    }
}
```

```swift
import Foundation

/// One line for the result list: a file and what happened to it.
public struct FileNote: Equatable, Sendable, Identifiable {
    public let name: String
    public let message: String
    /// The tool's own words, shown below the plain sentence.
    public let detail: String?

    public var id: String { name + "\n" + message }

    public init(name: String, message: String, detail: String?) {
        self.name = name
        self.message = message
        self.detail = detail
    }
}

public struct RunSummary: Equatable, Sendable {
    public var converted = 0
    public var set = 0
    public var skipped = 0
    public var failed = 0
    public var notes: [FileNote] = []
    public var cancelled = false

    public init() {}
}

public struct RunProgress: Equatable, Sendable {
    public let completed: Int
    public let total: Int
    public let name: String
}

/// Set from the main thread, read from the job's thread.
public final class CancelFlag: @unchecked Sendable {
    private let lock = NSLock()
    private var value = false

    public init() {}

    public func cancel() {
        lock.lock()
        value = true
        lock.unlock()
    }

    public var isCancelled: Bool {
        lock.lock()
        defer { lock.unlock() }
        return value
    }
}

/// One run over a folder snapshot (design, section 4.3).
///
/// Order: convert every pending ARW, then tag every DNG, the new ones last.
/// One file at a time. Cancel is honored between files.
public struct DesqueezeJob: Sendable {
    private let tag: ScaleTagging
    private let converter: RawConverter?
    private let backups: BackupHandling

    /// - Parameter converter: nil when Adobe DNG Converter is not installed.
    ///   The DNG files still run.
    public init(tag: ScaleTagging, converter: RawConverter?, backups: BackupHandling = FileBackups()) {
        self.tag = tag
        self.converter = converter
        self.backups = backups
    }

    /// - Parameter convert: false for Remove, which never converts.
    public func run(
        _ snapshot: FolderSnapshot,
        factor: SqueezeFactor,
        convert: Bool,
        cancel: CancelFlag = CancelFlag(),
        progress: (RunProgress) -> Void = { _ in }
    ) -> RunSummary {
        var summary = RunSummary()
        guard snapshot.isWritable else {
            summary.notes.append(FileNote(name: snapshot.folder.lastPathComponent,
                                          message: "This folder is read-only. No file was changed.", detail: nil))
            return summary
        }

        let arws = (convert && converter != nil) ? snapshot.pendingARWs : []
        var dngs = snapshot.dngs
        var total = arws.count * 2 + dngs.count
        var completed = 0

        for arw in arws {
            if cancel.isCancelled {
                summary.cancelled = true
                return summary
            }
            do {
                let dng = try converter!.convert(arw)
                summary.converted += 1
                dngs.append(DNGFile(url: dng, horizontal: 1, vertical: 1,
                                    originalRawFileName: arw.lastPathComponent, hasBackup: false))
            } catch {
                summary.failed += 1
                total -= 1
                summary.notes.append(FileNote(name: arw.lastPathComponent, message: "Not converted.",
                                              detail: Self.describe(error)))
            }
            completed += 1
            progress(RunProgress(completed: completed, total: total, name: arw.lastPathComponent))
        }

        for dng in dngs {
            if cancel.isCancelled {
                summary.cancelled = true
                return summary
            }
            tagOne(dng, factor, &summary)
            completed += 1
            progress(RunProgress(completed: completed, total: total, name: dng.name))
        }
        return summary
    }

    private func tagOne(_ dng: DNGFile, _ factor: SqueezeFactor, _ summary: inout RunSummary) {
        if factor.matches(horizontal: dng.horizontal, vertical: dng.vertical) {
            summary.skipped += 1
            return
        }
        // exiftool never replaces an existing `_original`. A write over one
        // would leave a backup of the wrong state, and this app must not
        // discard a backup that it did not make.
        if backups.backupExists(for: dng.url) {
            summary.skipped += 1
            summary.notes.append(FileNote(
                name: dng.name,
                message: "Skipped. An older backup, \(dng.name)_original, is beside this file. Move it away, then run again.",
                detail: nil))
            return
        }

        do {
            try tag.write(factor, to: dng.url)
            guard let now = try tag.read(dng.url),
                  factor.matches(horizontal: now.horizontal, vertical: now.vertical) else {
                throw TagError.toolFailed("The tag did not read back as \(factor.tagValue).")
            }
        } catch {
            // A failed write makes no backup and leaves the file unchanged.
            // Restore only when a backup exists.
            var message = "Not changed."
            if backups.backupExists(for: dng.url) {
                do {
                    try backups.restoreBackup(for: dng.url)
                    message = "Not changed. The backup was restored."
                } catch {
                    message = "The write failed, and the backup could not be restored. The backup is \(dng.name)_original."
                }
            }
            summary.failed += 1
            summary.notes.append(FileNote(name: dng.name, message: message, detail: Self.describe(error)))
            return
        }

        summary.set += 1
        do {
            try backups.trashBackup(for: dng.url)
        } catch {
            summary.notes.append(FileNote(
                name: dng.name,
                message: "Done. The backup stayed beside the file, because the Trash refused it.",
                detail: error.localizedDescription))
        }
    }

    static func describe(_ error: Error) -> String {
        switch error {
        case TagError.toolFailed(let text): return text
        case ConvertError.failed(let text): return text
        default: return error.localizedDescription
        }
    }
}
```

```swift
import AppKit
import DesqueezeCore
import SwiftUI

/// The window's state. All file work is in `DesqueezeCore`. This type owns
/// the folder, the factor, and the one running job.
@MainActor
final class AppModel: ObservableObject {
    enum Choice: Equatable {
        case preset(SqueezeFactor)
        case custom
    }

    @Published private(set) var snapshot: FolderSnapshot?
    @Published private(set) var folderMessage: String?
    @Published var choice: Choice
    @Published var customText: String
    @Published var customError: String?
    @Published private(set) var progress: RunProgress?
    @Published private(set) var summary: RunSummary?
    @Published private(set) var isRunning = false

    let exiftool: URL?
    let converter: URL?

    private var cancelFlag: CancelFlag?
    private static let factorKey = "lastFactor"

    init() {
        let registered = NSWorkspace.shared.urlForApplication(withBundleIdentifier: ToolLocator.converterBundleID)
        let locator = ToolLocator(registeredConverterApp: registered)
        // `-simulateMissingTools YES` on the command line shows the window as
        // it is on a Mac with neither tool. For checks by hand only.
        let simulateMissing = UserDefaults.standard.bool(forKey: "simulateMissingTools")
        exiftool = simulateMissing ? nil : locator.exiftool()
        converter = simulateMissing ? nil : locator.dngConverter()

        let saved = UserDefaults.standard.string(forKey: Self.factorKey).flatMap { SqueezeFactor(text: $0) }
            ?? SqueezeFactor(1.5)!
        if SqueezeFactor.presets.contains(saved) {
            choice = .preset(saved)
            customText = ""
        } else {
            choice = .custom
            customText = saved.label
        }
    }

    // MARK: What the view reads

    var factor: SqueezeFactor? {
        switch choice {
        case .preset(let preset): return preset
        case .custom: return SqueezeFactor(text: customText)
        }
    }

    var folderName: String? { snapshot?.folder.lastPathComponent }

    /// For example `216 DNG · 12 ARW to convert`.
    var countsLine: String? {
        guard let snapshot else { return nil }
        if snapshot.isEmpty { return "No DNG or ARW files in this folder." }
        var parts = ["\(snapshot.dngs.count) DNG"]
        if !snapshot.pendingARWs.isEmpty {
            parts.append(converter == nil
                ? "\(snapshot.pendingARWs.count) ARW cannot be converted: Adobe DNG Converter is not installed"
                : "\(snapshot.pendingARWs.count) ARW to convert")
        }
        return parts.joined(separator: " · ")
    }

    /// The count in the run button: DNG files, plus ARW files that can convert.
    var runCount: Int {
        guard let snapshot else { return 0 }
        return snapshot.dngs.count + (converter == nil ? 0 : snapshot.pendingARWs.count)
    }

    var canRun: Bool {
        exiftool != nil && !isRunning && runCount > 0 && snapshot?.isWritable == true
    }

    // MARK: Folder

    func chooseFolder() {
        let panel = NSOpenPanel()
        panel.canChooseDirectories = true
        panel.canChooseFiles = false
        panel.allowsMultipleSelection = false
        panel.prompt = "Choose"
        if panel.runModal() == .OK, let url = panel.url { load(url) }
    }

    func handleDrop(_ urls: [URL]) -> Bool {
        guard !isRunning else { return false }
        guard urls.count == 1, let url = urls.first,
              (try? url.resourceValues(forKeys: [.isDirectoryKey]).isDirectory) == true else {
            folderMessage = "Drop a folder"
            return false
        }
        load(url)
        return true
    }

    func load(_ folder: URL) {
        summary = nil
        progress = nil
        scan(folder)
    }

    private func scan(_ folder: URL) {
        folderMessage = nil
        guard let exiftool else {
            snapshot = nil
            return
        }
        let scanner = FolderScanner(tag: ScaleTag(exiftool: exiftool))
        Task {
            do {
                let result = try await Task.detached { try scanner.snapshot(of: folder) }.value
                snapshot = result
                if !result.isWritable {
                    folderMessage = "This folder is read-only. Desqueeze cannot change its files."
                }
            } catch {
                snapshot = nil
                folderMessage = "This folder could not be read."
            }
        }
    }

    // MARK: Run

    func run() {
        guard let factor else {
            customError = "Enter a number from 1 to 4, with at most four decimal places."
            return
        }
        customError = nil
        UserDefaults.standard.set(factor.label, forKey: Self.factorKey)
        start(factor: factor, convert: true)
    }

    func remove() {
        start(factor: .noStretch, convert: false)
    }

    func cancel() {
        cancelFlag?.cancel()
    }

    private func start(factor: SqueezeFactor, convert: Bool) {
        guard !isRunning, let snapshot, let exiftool else { return }
        let job = DesqueezeJob(
            tag: ScaleTag(exiftool: exiftool),
            converter: converter.map { AdobeConverter(executable: $0) }
        )
        let flag = CancelFlag()
        cancelFlag = flag
        isRunning = true
        summary = nil
        progress = nil

        Task {
            let result = await Task.detached {
                job.run(snapshot, factor: factor, convert: convert, cancel: flag) { update in
                    Task { @MainActor in self.progress = update }
                }
            }.value
            summary = result
            isRunning = false
            cancelFlag = nil
            scan(snapshot.folder)
        }
    }
}
```

````markdown
## Invariants

1. **The original is never at risk.** The app never changes, moves, or deletes an ARW. A DNG keeps its `_original` backup until the tag reads back correctly.
2. **No pixels change.** The app sets a tag. It never resamples or exports an image.
3. **The app does not touch a backup it did not make.** A DNG with an existing `_original` is skipped. exiftool never replaces one, so a write there would leave a backup of the wrong state.
4. **Only a pending ARW goes to the converter.** When the DNG exists, Adobe DNG Converter writes a `_1.dng` duplicate.
5. **A conversion passes only when the exit code is 0 and the DNG exists.** The converter returns 0 for a file it cannot read.
6. **Never call Adobe DNG Converter with `-h`.** It opens a window and never returns.
7. **Top level only.** The scan does not enter subfolders. They hold exports and Capture One data.
8. **Every exiftool call names `SubIFD`.** That is the directory Camera Raw reads.
9. **The app is not sandboxed.** No entitlements file exists, on purpose. A sandboxed app cannot start the two tools.
10. **`DesqueezeCore` imports Foundation only.** AppKit and SwiftUI stay in `App/`. All file work stays in the package, behind protocols, so tests use fakes.
11. **Offline and private.** No network requests, no analytics.
12. **UI copy:** sentence case, no exclamation marks, no emoji. An error is one plain sentence, with the tool's own words on a detail line below it.
````
